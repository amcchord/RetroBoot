cmake_minimum_required(VERSION 3.10)
project(RetroBootColor C)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(RetroPPC|Retro68)$")
    message(FATAL_ERROR "Use Retro68's PowerPC or 68k toolchain")
endif()
if(NOT RB_CONFIG_RESOURCE OR NOT RB_CREATOR)
    message(FATAL_ERROR "Use build.py to provide the original name and creator")
endif()
set(CMAKE_C_STANDARD 99)
add_compile_options(-O2 -Wall -Wextra -Wno-multichar)
add_application(RetroBootColor TYPE APPL CREATOR ${RB_CREATOR}
    FILES color.c color.r "${RB_CONFIG_RESOURCE}")
if(CMAKE_SYSTEM_NAME STREQUAL "RetroPPC")
    target_link_libraries(RetroBootColor AppearanceLib)
endif()
