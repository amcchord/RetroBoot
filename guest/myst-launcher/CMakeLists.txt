cmake_minimum_required(VERSION 3.10)
project(RetroBootMyst C)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^RetroPPC$")
    message(FATAL_ERROR "Use Retro68's PowerPC toolchain")
endif()
set(CMAKE_C_STANDARD 99)
add_compile_options(-O2 -Wall -Wextra -Wno-multichar)
add_application(RetroBootMyst TYPE APPL CREATOR RBmy FILES myst.c myst.r)
target_link_libraries(RetroBootMyst AppearanceLib)
